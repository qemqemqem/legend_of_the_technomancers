% ---- Sheet-local helpers --------------------------------------------

\newcommand{\fivepips}{%
  $\vcenter{\hbox{%
    \begin{tikzpicture}[x=9pt,y=9pt]
      \foreach \i in {1,...,5}{%
        \node[circle,draw=atrRule,line width=0.4pt,minimum size=7pt,inner sep=0]
          at (\i,0){};}
    \end{tikzpicture}}}$}

\newcommand{\fourpips}{%
  $\vcenter{\hbox{%
    \begin{tikzpicture}[x=9pt,y=9pt]
      \foreach \i in {1,...,4}{%
        \node[circle,draw=atrRule,line width=0.4pt,minimum size=7pt,inner sep=0]
          at (\i,0){};}
    \end{tikzpicture}}}$}

\newcommand{\hptrack}{%
  $\vcenter{\hbox{%
    \begin{tikzpicture}[x=11pt,y=11pt]
      \foreach \i in {1,...,10}{%
        \node[circle,draw=atrRule,line width=0.5pt,minimum size=9pt,inner sep=0]
          at (\i,0){};}
    \end{tikzpicture}}}$}

% Tag oval for body tags
\newcommand{\tagslot}[1][38mm]{%
  $\vcenter{\hbox{%
    \begin{tikzpicture}
      \node[rounded rectangle, draw=atrRule, line width=0.5pt,
            minimum width=#1, minimum height=5.5mm, inner sep=3pt]{\phantom{X}};
    \end{tikzpicture}}}$}

% Pre-printed skill row: name | stat | pips
\newcommand{\skillrow}[2]{%
  {\small #1} & {\scriptsize\color{atrAccent}\textbf{#2}} & \fivepips\\[1.5pt]}

% Blank skill row: blank name | blank stat letter | pips
\newcommand{\skillblank}{%
  \rule{30mm}{0.3pt} &
  \raisebox{1pt}{\framebox[5mm]{\phantom{B}}} &
  \fivepips\\[1.5pt]}

% Item row for stuff
\newcommand{\itemrow}{\rule{42mm}{0.3pt}\hspace{3pt}\fourpips\par\vspace{4pt}}

% Coloured section heading
\newcommand{\sechead}[2][atrAccent]{%
  \par{\color{#1}\rule{\linewidth}{0.7pt}}\par\vspace{1pt}%
  {\hdfont\bfseries\footnotesize\color{#1}#2}\par\vspace{4pt}}

% ---- The tattoo drawing box -----------------------------------------
% White interior, double-border frame, invites drawing with markers.
\newcommand{\tattoobox}[2][55mm]{%
  % #1 = height, drawn via tikz so it is an exact size
  \par
  \begin{tikzpicture}
    % Outer border
    \draw[color=atrAccent, line width=0.9pt, rounded corners=2pt]
      (0,0) rectangle (\linewidth, #1);
    % Inner decorative border (inset 2mm)
    \draw[color=atrAccent, line width=0.3pt, opacity=0.5, rounded corners=1pt]
      (2mm, 2mm) rectangle (\linewidth-2mm, #1-2mm);
    % Very faint label at the bottom
    \node[anchor=south, opacity=0.18, font=\hdfont\footnotesize\color{atrAccent}]
      at (0.5\linewidth, 2.5mm) {TATTOOS};
  \end{tikzpicture}
  \par}
