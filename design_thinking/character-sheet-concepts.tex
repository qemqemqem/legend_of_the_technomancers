% ---- Helper commands ------------------------------------------------

% Underline fill of given width
\newcommand{\fillline}[1]{\hspace{2pt}\underline{\hspace{#1}}}

% Five pip row for stats (1-5 scale) — $\vcenter$ keeps it on the text baseline
\newcommand{\fivepips}{%
  $\vcenter{\hbox{%
    \begin{tikzpicture}[x=9pt,y=9pt]
      \foreach \i in {1,...,5}{%
        \node[circle,draw=atrRule,line width=0.4pt,minimum size=7pt,inner sep=0]
          at (\i,0){};}
    \end{tikzpicture}}}$}

% Four pip row for magic rank (1-4 scale)
\newcommand{\fourpips}{%
  $\vcenter{\hbox{%
    \begin{tikzpicture}[x=9pt,y=9pt]
      \foreach \i in {1,...,4}{%
        \node[circle,draw=atrRule,line width=0.4pt,minimum size=7pt,inner sep=0]
          at (\i,0){};}
    \end{tikzpicture}}}$}

% HP circles — 10 circles, larger
\newcommand{\hptrack}{%
  $\vcenter{\hbox{%
    \begin{tikzpicture}[x=11pt,y=11pt]
      \foreach \i in {1,...,10}{%
        \node[circle,draw=atrRule,line width=0.5pt,minimum size=9pt,inner sep=0]
          at (\i,0){};}
    \end{tikzpicture}}}$}

% A blank tag oval drawn with TikZ rounded-rectangle
\newcommand{\tagslot}[1][36mm]{%
  $\vcenter{\hbox{%
    \begin{tikzpicture}
      \node[rounded rectangle, draw=atrRule, line width=0.5pt,
            minimum width=#1, minimum height=5.5mm, inner sep=3pt]{\phantom{X}};
    \end{tikzpicture}}}$}

% One item row: name line + tier pips on one line
\newcommand{\itemrow}{\rule{44mm}{0.3pt}\hspace{3pt}\fourpips\par\vspace{4pt}}

% One skill row in a tabular: name | stat letter | pips
\newcommand{\skillrow}[2]{%
  {\small #1} & {\small\color{atrAccent}#2} & \fivepips\\[2pt]}

% Coloured section header with rule
\newcommand{\sechead}[2][atrAccent]{%
  \par{\color{#1}\rule{\linewidth}{0.7pt}}\par\vspace{1pt}%
  {\hdfont\bfseries\footnotesize\color{#1}#2}\par\vspace{3pt}}

% Tattoo drawing box — white interior, double-border frame, for drawing with markers
\newcommand{\tattoobox}[1][55mm]{%
  \par
  \begin{tikzpicture}
    \draw[color=atrAccent, line width=0.9pt, rounded corners=2pt]
      (0,0) rectangle (\linewidth, #1);
    \draw[color=atrAccent, line width=0.3pt, opacity=0.5, rounded corners=1pt]
      (2mm, 2mm) rectangle (\linewidth-2mm, #1-2mm);
    \node[anchor=south, opacity=0.18, font=\hdfont\footnotesize\color{atrAccent}]
      at (0.5\linewidth, 2.5mm) {TATTOOS};
  \end{tikzpicture}
  \par}

% Blank skill row: underline name | framebox for stat letter | pips
\newcommand{\skillblank}{%
  \rule{30mm}{0.3pt} &
  \raisebox{1pt}{\framebox[5mm]{\phantom{B}}} &
  \fivepips\\[1.5pt]}

% Framed stat block (for use inside minipages, not tabular cells)
% #1 = frame colour, #2 = icon/stat name (body, mind, face)
\newcommand{\statbox}[2]{%
  \begin{tcolorbox}[enhanced, colback=atrBoxbg, colframe=#1,
    boxrule=0.8pt, arc=2pt, left=4pt, right=4pt, top=4pt, bottom=4pt,
    halign=center]
    {\color{#1}\atricon{#2}\ \hdfont\small\MakeUppercase{#2}}\\[5pt]
    \fivepips
  \end{tcolorbox}}

% Big stat block with larger icon (for Concept B)
\newcommand{\bigstatbox}[2]{%
  \par\smallskip
  \begin{center}
  \begin{tcolorbox}[enhanced, colback=atrBoxbg, colframe=#1,
    boxrule=1pt, arc=3pt, left=6pt, right=6pt, top=6pt, bottom=6pt,
    halign=center, width=0.9\linewidth]
    {\color{#1}{\setlength{\atrIconH}{20pt}\atricon{#2}}}\\[4pt]
    {\hdfont\normalsize\color{#1}\MakeUppercase{#2}}\\[6pt]
    \fivepips
  \end{tcolorbox}
  \end{center}}
