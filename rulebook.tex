% ------------------------------------------------- Full-bleed cover page
% The art fills the entire page; title and authors are overlaid on scrims
% (tikz overlay on current page = ignores margins, bleeds to the edges).
\begin{titlingpage}
% Black page background: any gap between image and physical paper edge
% shows as black rather than white, matching the dark cover art.
\pagecolor{black}
% Single value controls author band height; text is automatically centred in it.
\newlength{\atrAuthBandH}\setlength{\atrAuthBandH}{54mm}
\begin{tikzpicture}[remember picture, overlay]
  % "Cover" fit, not "contain": the image is deliberately oversized
  % (1.08x paperwidth) and hard-clipped to the page box, so it always
  % fills every edge regardless of the source aspect ratio. Anchoring
  % south-west keeps the crop attached to the bottom-left corner, so
  % overflow is trimmed from the top and right only. Plain
  % width=\paperwidth measured ~5.7mm short of the right edge in
  % practice (the source PNG itself has no such gap) — the clip plus
  % overscan makes the fit robust to that slack instead of exact-fitting
  % against it. \atrpagecorners (atrpg.sty) works around a separate bug
  % where "current page.south *" itself is ~50bp off the true page edge
  % under this toolchain — see the comment there.
  \atrpagecorners
  \clip (atrpagesw) rectangle (current page.north east);
  \node[anchor=south west, inner sep=0] at (atrpagesw)
    {\includegraphics[width=1.08\paperwidth]{art/cover-full.png}};

  % ---- Title band (top) ----
  \fill[black, opacity=0.42]
    (current page.north west) rectangle ([yshift=-94mm]current page.north east);
  \node[anchor=north, align=center, inner sep=0] at ([yshift=-26mm]current page.north){%
    {\hdfont\bfseries\color{white}\fontsize{44}{46}\selectfont LEGEND OF THE}\\[3pt]
    {\hdfont\bfseries\color{white}\fontsize{44}{46}\selectfont TECHNOMANCERS}\\[7mm]
    {\color{white}\rule{0.52\paperwidth}{0.8pt}}\\[6mm]
    {\hdfont\large\itshape\color{white}A Roleplaying Game for Traveling Across Realities}%
  };

  % ---- Author band (bottom) — change \atrAuthBandH above to resize. ----
  \fill[black, opacity=0.42]
    ([yshift=\atrAuthBandH]atrpagesw) rectangle (atrpagese);
  \node[anchor=center, align=center, inner sep=0]
    at ([yshift=\dimexpr\atrAuthBandH/2\relax]atrpages){%
    {\hdfont\large\color{white}Andrew Keenan Richardson}\\[3pt]
    {\hdfont\normalsize\color{white}\&}\\[3pt]
    {\hdfont\large\color{white}Christopher Sheldon-Dante}\\[3pt]
    {\hdfont\normalsize\color{white}\&}\\[3pt]
    {\hdfont\large\color{white}Anthropic Claude}%
  };
\end{tikzpicture}
\end{titlingpage}
\pagecolor{white}  % reset after the black cover background

\frontmatter
% One column, Parts + Chapters only (tocdepth set above): a short list
% like this reads better as a single deliberate page than split across
% two half-empty columns. Styling lives in atrpg.sty ("Table of Contents
% styling"). Bypassing \tableofcontents itself (calling \chapter* and
% \@starttoc directly, the same machinery it uses internally) rather
% than calling it plainly: hyperref patches \tableofcontents to add a
% self-referential "Contents" entry (for its own PDF bookmark) that then
% shows up as the first line of the printed list — \chapter* alone
% doesn't do that.
\makeatletter
\chapter*{\contentsname}
\@starttoc{toc}
\makeatother
