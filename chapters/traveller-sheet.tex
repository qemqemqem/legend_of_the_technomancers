% ---- Sheet-local helper commands ------------------------------------

\newcommand{\sheetfivepips}{%
  $\vcenter{\hbox{%
    \begin{tikzpicture}[x=9pt,y=9pt]
      \foreach \i in {1,...,5}{%
        \node[circle,draw=atrRule,line width=0.4pt,minimum size=7pt,inner sep=0]
          at (\i,0){};}
    \end{tikzpicture}}}$}

\newcommand{\sheetfourpips}{%
  $\vcenter{\hbox{%
    \begin{tikzpicture}[x=9pt,y=9pt]
      \foreach \i in {1,...,4}{%
        \node[circle,draw=atrRule,line width=0.4pt,minimum size=7pt,inner sep=0]
          at (\i,0){};}
    \end{tikzpicture}}}$}

\newcommand{\sheettagslot}[1][38mm]{%
  $\vcenter{\hbox{%
    \begin{tikzpicture}
      \node[rounded rectangle, draw=atrRule, line width=0.5pt,
            minimum width=#1, minimum height=5.5mm, inner sep=3pt]{\phantom{X}};
    \end{tikzpicture}}}$}

% A skill is a boolean tag (known or not) --- one checkbox, not a rating.
\newcommand{\sheetknownbox}{%
  $\vcenter{\hbox{%
    \begin{tikzpicture}
      \node[draw=atrRule, line width=0.4pt, minimum size=7pt, inner sep=0]{};
    \end{tikzpicture}}}$}

% A Stat's pool (Body/Mind/Face) --- five squares, cross one off per
% point lost. Squares (not circles) so it's never confused with the
% five-pip Stat rating directly above it on the sheet.
\newcommand{\sheetpooltrack}{%
  $\vcenter{\hbox{%
    \begin{tikzpicture}[x=9pt,y=9pt]
      \foreach \i in {1,...,5}{%
        \node[draw=atrRule,line width=0.4pt,minimum size=7pt,inner sep=0]
          at (\i,0){};}
    \end{tikzpicture}}}$}

\newcommand{\sheetskillrow}[2]{%
  {\small #1} & {\scriptsize\color{atrAccent}\textbf{#2}} & \sheetknownbox\\[1.5pt]}

\newcommand{\sheetskillblank}{%
  \rule{30mm}{0.3pt} &
  \raisebox{1pt}{\framebox[5mm]{\phantom{B}}} &
  \sheetknownbox\\[1.5pt]}

\newcommand{\sheetitemrow}{\rule{42mm}{0.3pt}\hspace{3pt}\sheetfourpips\par\vspace{4pt}}

\newcommand{\sheetsechead}[2][atrAccent]{%
  \par{\color{#1}\rule{\linewidth}{0.7pt}}\par\vspace{1pt}%
  {\hdfont\bfseries\footnotesize\color{#1}#2}\par\vspace{4pt}}

\newcommand{\sheettattoobox}[1][55mm]{%
  \par
  \begin{tikzpicture}
    \draw[color=atrAccent, line width=0.9pt, rounded corners=2pt]
      (0,0) rectangle (\linewidth, #1);
    \draw[color=atrAccent, line width=0.3pt, opacity=0.5, rounded corners=1pt]
      (2mm, 2mm) rectangle (\linewidth-2mm, #1-2mm);
    \node[anchor=south, opacity=0.18, font=\hdfont\footnotesize\color{atrAccent}]
      at (0.5\linewidth, 2.5mm) {TATTOOS};
  \end{tikzpicture}
  \par}
